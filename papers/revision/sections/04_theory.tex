\subsection{Root geometry and exact nonsubsampled tightness}

\begin{proposition}[Mapped geometry and phase localization; exact]
\label{prop:phase-pole}
Let $|\alpha|<1$.  For real $\lambda$,
$|B_\alpha(\cayley(\lambda))|=1$, and
\begin{equation}
 B_\alpha(\cayley(\lambda))
 =\frac{(1-\alpha)\lambda-i(1+\alpha)}
 {(1-\overline\alpha)\lambda+i(1+\overline\alpha)}.
 \label{eq:rational-form}
\end{equation}
Its mapped zero and pole are
\begin{equation}
 z_\alpha=i\frac{1+\alpha}{1-\alpha}=a+ib,
 \qquad
 p_\alpha=\overline{z_\alpha}=a-ib,
 \qquad b>0.
 \label{eq:mapped-zero-pole}
\end{equation}
For a continuous phase branch,
\begin{equation}
 \frac{d}{d\lambda}\arg B_\alpha(\cayley(\lambda))
 =\frac{1-|\alpha|^2}{|\cayley(\lambda)-\alpha|^2}
   \frac{2}{1+\lambda^2}
 =\frac{2b}{(\lambda-a)^2+b^2}>0.
 \label{eq:phase-law}
\end{equation}
If $\alpha=\rho e^{i\vartheta}$, then
\begin{equation}
 a=\frac{-2\rho\sin\vartheta}{1-2\rho\cos\vartheta+\rho^2},
 \qquad
 b=\frac{1-\rho^2}{1-2\rho\cos\vartheta+\rho^2}.
 \label{eq:radial-mapped-geometry}
\end{equation}
For a product of factors, the phase derivatives in \cref{eq:phase-law} add.
Consequently the optional choice in \cref{eq:center-width-root}, with
$\mu\in[0,2]$ and frozen
$0<\gamma_{\min}\leq\gamma\leq\gamma_{\max}<\infty$, has an in-spectrum
phase-derivative center $\mu$, half-width at half maximum $\gamma$, and target
pole margin $\gamma$.  For an unrestricted radial root, $a$ is the real-axis
center; if $a\notin[0,2]$, the maximum observed on the graph interval occurs
at the nearer endpoint.  A radial angle anchor is not an exact center.
\end{proposition}

\begin{theorem}[T-A: pointwise multilevel partition and exact split]
\label{thm:pointwise-partition}
\label{thm:tight-split}
Let $L$ be self-adjoint and let all level roots be fixed in $\D$.  Write
$t_\ell(\lambda)=B_{\mathcal R_\ell}(\cayley(\lambda))$,
$p_\ell^\pm=(1\pm t_\ell)/2$, and define the effective atoms
\begin{equation}
 a_\ell(\lambda)=p_\ell^-(\lambda)
                  \prod_{j<\ell}p_j^+(\lambda)\quad(0\leq\ell<D),
 \qquad
 a_D(\lambda)=\prod_{j<D}p_j^+(\lambda).
 \label{eq:effective-atoms}
\end{equation}
The scalar multilevel analysis vector
$a(\lambda)=(a_0(\lambda),\ldots,a_D(\lambda))^\top$ obeys the pointwise
paraunitary identity
\begin{equation}
 \sum_{\ell=0}^{D}|a_\ell(\lambda)|^2=1
 \quad(\lambda\in\R).
 \label{eq:pointwise-partition}
\end{equation}
At each level, $T_\ell=t_\ell(L)$ is unitary and
\begin{equation}
 (P_\ell^-)^*P_\ell^-+(P_\ell^+)^*P_\ell^+=I,
 \qquad
 \norm{P_\ell^-h}_F^2+\norm{P_\ell^+h}_F^2=\norm h_F^2.
 \label{eq:parseval-operator}
\end{equation}
The bank is nonsubsampled and redundant; this identity makes no
critical-sampling claim.  The carried operator satisfies
$\norm{P_\ell^+}_{\op}\leq1$.
\end{theorem}

\begin{proposition}[Additive versus adjoint reconstruction]
\label{prop:two-reconstructions}
At each level, additive reconstruction
$r_\ell+h_{\ell+1}=h_\ell$ holds whenever the same exact or approximate
factor defines both channels.  For the exact bank, the terminal condition and
backward recursion in \cref{eq:synthesis-recursion} give
$h_0^{\mathrm{syn}}=h_0$.  The separately defined approximate channels and
recursion in \cref{eq:approx-synthesis-recursion} return
$\widetilde\Acal_D^*\widetilde\Acal_D\widetilde h_0$ and are not generally an
inverse.
\end{proposition}

\begin{theorem}[T-C: exact multilevel isometry and conditioning]
\label{thm:multilevel-pr}
For fixed roots and exact factors at $D$ levels,
\begin{equation}
 \norm{\Acal_Dh_0}_{\oplus,F}^2
 =\sum_{\ell=0}^{D-1}\norm{r_\ell}_F^2+\norm{h_D}_F^2
 =\norm{h_0}_F^2,
 \qquad \Acal_D^*\Acal_D=I.
 \label{eq:multilevel-isometry}
\end{equation}
Thus every singular value of the rectangular analysis map on its input domain
equals one, $\norm{\Acal_D\,\delta h}_{\oplus,F}=\norm{\delta h}_F$, and
$\kappa(\Acal_D)=1$.  In general
$\Acal_D\Acal_D^*\neq I$ because the coefficient space is redundant.  Adjoint
synthesis reconstructs every analyzed signal.
\end{theorem}

Weighted spectral Parseval conservation, the additive node-pair bound, and
spectral selection/recovery inequalities are useful consequences rather than
the central claim chain.  We state them in
\cref{thm:weighted-parseval,eq:nodewise-lower,thm:energy-separation,cor:complex-recovery}
in \cref{app:proofs}.  In particular, the node-pair inequality is a weak
redundancy bound and is not presented as resistance to representation
collapse.

\subsection{Finite approximation and target-pole complexity}

\begin{theorem}[T-E: Chebyshev interpolation under target analyticity]
\label{thm:pole-distance-cheb}
Let the nonconstant exact rational target
$g(\lambda)=B_{\mathcal R}(\cayley(\lambda))$ have a nonempty multiset
$\mathcal P_{\mathrm{red}}(g)$ of \emph{reduced nonremovable target poles}.
For $p\in\mathcal P_{\mathrm{red}}(g)$ set $\xi_p=p-1$ and define
\begin{equation}
 \chi(\xi_p)=\max_{\pm}|\xi_p\pm\sqrt{\xi_p^2-1}|.
\end{equation}
For $1<\varrho<\min_{p\in\mathcal P_{\mathrm{red}}(g)}\chi(\xi_p)$, let
$E_\varrho$ be the closed Bernstein ellipse for $[-1,1]$ and let
$M_\varrho=\max_{x\in E_\varrho}|g(1+x)|$.  The degree-$K$ interpolant $p_K$
at the $K+1$ first-kind nodes
$\lambda_j=1+\cos((j+\tfrac12)\pi/(K+1))$, $j=0,\ldots,K$, satisfies
\begin{equation}
 \sup_{\lambda\in[0,2]}|g(\lambda)-p_K(\lambda)|
 \leq \frac{4M_\varrho}{\varrho-1}\varrho^{-K}.
 \label{eq:cheb-bound}
\end{equation}
For self-adjoint $L$ the same bound holds in operator norm.  Hence this upper
bound is at most $\varepsilon$ whenever
\begin{equation}
 K\geq
 \left\lceil\frac{\log(4M_\varrho/((\varrho-1)\varepsilon))}
 {\log\varrho}\right\rceil_+.
 \label{eq:cheb-degree}
\end{equation}
Here $\lceil x\rceil_+=\max\{0,\lceil x\rceil\}$.
Both the admissible ellipse and $M_\varrho$ are essential: target-pole distance
alone does not determine the realized interpolation error.  The poles belong
to the reduced rational target; the finite interpolant itself is a polynomial
without finite poles.
\end{theorem}

\begin{theorem}[T-D: finite-Chebyshev multilevel frame distortion]
\label{thm:finite-frame}
Suppose the level-$\ell$ polynomial factor obeys the \emph{true} error
$\norm{\widetilde T_\ell-T_\ell}_{\op}\leq\epsilon_\ell$.  Define
\begin{equation}
 d_\ell=\epsilon_\ell+\frac{\epsilon_\ell^2}{2},
 \qquad c_\ell=\left(1+\frac{\epsilon_\ell}{2}\right)^2,
 \qquad
 \Delta_D=\sum_{\ell=0}^{D-1}d_\ell
              \prod_{j=0}^{\ell-1}c_j.
 \label{eq:finite-delta}
\end{equation}
For the complete approximate analysis map $\widetilde\Acal_D$,
\begin{equation}
 \norm{\widetilde\Acal_D^*\widetilde\Acal_D-I}_{\op}\leq\Delta_D.
 \label{eq:finite-frame-distortion}
\end{equation}
If $\Delta_D<1$, valid frame bounds (equivalently, squared singular-value
bounds) are $1-\Delta_D$ and $1+\Delta_D$.  No positive lower frame bound is
asserted from this estimate when $\Delta_D\geq1$.  Additive reconstruction remains exact for coefficients
formed by the same approximate channels.  For coefficients obtained by
analyzing $h$, approximate adjoint synthesis has error
$\norm{(\widetilde\Acal_D^*\widetilde\Acal_D-I)h}_F
\leq\Delta_D\norm h_F$; this is not an inverse guarantee for arbitrary
coefficient tuples.
\end{theorem}

\subsection{What the exact isometry does not guarantee}

\begin{proposition}[Negative boundary for oversmoothing and oversquashing]
\label{prop:sensitivity-boundary}
With fixed roots, the Jacobian of the complete exact coefficient map is
$\Acal_D$, so its \emph{global} singular values equal one.  Nevertheless no
positive, graph-independent lower bound follows for a prescribed
source-to-target Jacobian block.  Indeed, for $D=1$, a real root $r\uparrow1$
gives $B_r(\cayley(L))\to-I$ in operator norm; for any $u\neq v$, the block
mapping a perturbation at $u$ to both coefficients at $v$ therefore tends to
zero while the complete map remains isometric.  It is exactly zero across
disconnected components.  For a Chebyshev-$K$ depth-$D$ realization it is
also exactly zero when $\operatorname{dist}(u,v)>DK$.

Moreover, for any real root, $B_r(\cayley(0))=-1$, so the zero-eigenspace is
annihilated by the carried $P^+$ branch in one level even though it is retained
by the emitted coefficient.  Complete coefficients and carried state must
therefore be evaluated separately.  Parseval tightness, invertibility,
energy conservation, and global Jacobian norm preservation alone establish
neither resistance to oversmoothing nor mitigation of oversquashing.
\end{proposition}
